## **Exercice/ Méthode 93 — Résolution graphique d'une équation du second degré**

```python
import numpy as np
import matplotlib.pyplot as plt

def resoudre_et_tracer(a, b, c):
    if a == 0:
        raise ValueError("Le coefficient 'a' doit être non nul.")

    delta = b**2 - 4 * a * c
    sommet_x = -b / (2 * a)
    sommet_y = c - (b**2) / (4 * a)

    print(f"\n--- Équation : {a}x² + ({b})x + ({c}) = 0 ---")
    print(f"Discriminant Δ = {delta}")

    racines = []
    if delta > 0:
        x1 = (-b - np.sqrt(delta)) / (2 * a)
        x2 = (-b + np.sqrt(delta)) / (2 * a)
        racines = [x1, x2]
        print("Interprétation : Δ > 0. L'équation admet deux solutions réelles distinctes.")
        print(f"Racines : x₁ = {x1:.3f}, x₂ = {x2:.3f}")
        print("Graphique : La parabole coupe l'axe des abscisses en deux points.")
    elif delta == 0:
        x0 = -b / (2 * a)
        racines = [x0]
        print("Interprétation : Δ = 0. L'équation admet une racine double.")
        print(f"Racine double : x₀ = {x0:.3f}")
        print("Graphique : Le sommet de la parabole est tangent à l'axe des abscisses.")
    else:
        print("Interprétation : Δ < 0. Aucune solution réelle (deux racines complexes conjuguées).")
        print("Graphique : La parabole ne coupe jamais l'axe des abscisses.")

    ecart = max(abs(r - sommet_x) for r in racines) + 2 if racines else 3
    x = np.linspace(sommet_x - ecart, sommet_x + ecart, 500)
    y = a * x**2 + b * x + c

    plt.figure(figsize=(8, 5))
    plt.plot(x, y, label=f"f(x) = {a}x² + {b}x + {c}", color="navy", lw=2)
    plt.axhline(0, color="black", linestyle="--", linewidth=1, alpha=0.7)
    plt.axvline(0, color="black", linestyle="--", linewidth=1, alpha=0.7)

    for i, r in enumerate(racines, 1):
        plt.scatter([r], [0], color="crimson", zorder=5, s=60, label=f"Racine {i} ({r:.2f}, 0)")

    plt.scatter([sommet_x], [sommet_y], color="darkorange", zorder=4, s=40, label=f"Sommet ({sommet_x:.2f}, {sommet_y:.2f})")

    plt.title(f"Résolution graphique : Δ = {delta}")
    plt.xlabel("x")
    plt.ylabel("f(x)")
    plt.grid(True, linestyle=":", alpha=0.6)
    plt.legend()
    plt.show()

# Tests
resoudre_et_tracer(a=1, b=2, c=-3)
resoudre_et_tracer(a=1, b=-4, c=4)
resoudre_et_tracer(a=1, b=2, c=5)
```

## **Exercice/ Méthode 96 — Forme factorisée d'un polynôme du second degré**

```python
import numpy as np

def forme_factorisee(a, b, c):
    if a == 0:
        raise ValueError("Le coefficient 'a' doit être non nul.")

    delta = b**2 - 4 * a * c
    print(f"\n--- Polynôme : {a}x² + ({b})x + ({c}) ---")
    print(f"Discriminant Δ = {delta}")

    if delta > 0:
        x1 = (-b - np.sqrt(delta)) / (2 * a)
        x2 = (-b + np.sqrt(delta)) / (2 * a)
        f1 = f"(x - {x1:.2f})" if x1 >= 0 else f"(x + {-x1:.2f})"
        f2 = f"(x - {x2:.2f})" if x2 >= 0 else f"(x + {-x2:.2f})"
        coef = "" if a == 1 else ("-" if a == -1 else f"{a}")
        print(f"Forme factorisée : P(x) = {coef}{f1}{f2}")
    elif delta == 0:
        x0 = -b / (2 * a)
        f0 = f"(x - {x0:.2f})" if x0 >= 0 else f"(x + {-x0:.2f})"
        coef = "" if a == 1 else ("-" if a == -1 else f"{a}")
        print(f"Forme factorisée : P(x) = {coef}{f0}²")
    else:
        print("Interprétation : Δ < 0. Pas de factorisation dans ℝ.")

# Tests
forme_factorisee(a=2, b=-5, c=3)
forme_factorisee(a=1, b=-4, c=4)
forme_factorisee(a=3, b=2, c=1)
```

## **Exercice/ Méthode 97 — Isolation des paramètres d'une fonction polynomiale**

### **1. Isolation des paramètres**

- **(a)** \(a = P(1) - b - c\)
- **(b)** \(b = a + c - P(-1)\)
- **(c)** \(c = P(0)\)

---

### **2. Application numérique**

- \(c = 1\)
- \(a = 2\)
- \(b = 1\)

\(P(x) = 2x^2 + x + 1\)

---

### **3. Programmation Python**

```python
def calcul_a(P1, b, c):
    return P1 - b - c

def calcul_b(P_moins_1, a, c):
    return a + c - P_moins_1

def trouver_abc(P1, P_moins_1, P0):
    c = P0
    a = (P1 + P_moins_1 - 2 * c) / 2
    b = P1 - a - c
    return a, b, c

a, b, c = trouver_abc(P1=4, P_moins_1=2, P0=1)
print(f"a = {a}, b = {b}, c = {c}")
```

---

### **4. Analyse graphique**

```python
import numpy as np
import matplotlib.pyplot as plt

a, b, c = 2, -3, 1

def P(x):
    return a * x**2 + b * x + c

x = np.linspace(-2, 2, 400)
y = P(x)

fig, ax = plt.subplots(figsize=(8, 5))
ax.plot(x, y, color="blue", label=f"P(x) = {a}x² + ({b})x + {c}")

points = [(-1, P(-1)), (0, P(0)), (1, P(1))]
for px, py in points:
    ax.plot(px, py, marker="o", color="red")
    ax.annotate(f"({px}, {py})", (px, py), textcoords="offset points", xytext=(-15, 10))

ax.axhline(0, color="black", linestyle="--", linewidth=1, alpha=0.7)
ax.axvline(0, color="black", linestyle="--", linewidth=1, alpha=0.7)
ax.set_title("Analyse graphique de P(x)")
ax.set_xlabel("x")
ax.set_ylabel("P(x)")
ax.grid(True)
ax.legend()

plt.show()
```

## **Exercice/ Méthode 103 — Trajectoire d'une balle**

### **1. Équations différentielles du mouvement**

$$
\begin{cases}
x''(t) = 0 \\
y''(t) = -g
\end{cases}
$$

---

### **2. Expressions de \(x(t)\) et \(y(t)\)**

$$
\begin{cases}
x(t) = v_{x0} \cdot t + x_0 \\
y(t) = -\frac{1}{2}gt^2 + v_{y0} \cdot t + y_0
\end{cases}
$$

---

### **3. Équation de la trajectoire \(y = ax^2 + bx + c\)**

$$
y(x) = -\frac{g}{2 v_{x0}^2} x^2 + \frac{v_{y0}}{v_{x0}} x
$$

- $a = -\frac{g}{2 v_{x0}^2}$
- $b = \frac{v*{y0}}{v*{x0}}$
- \(c = 0\)

---

### **4. Fonctions Python**

```python
import numpy as np
import matplotlib.pyplot as plt

def trajectoire_balle(vx0, vy0, x0=0, y0=0, g=9.81):
    a = -g / (2 * vx0**2)
    b = (vy0 / vx0) - (2 * a * x0)
    c = a * x0**2 - (vy0 / vx0) * x0 + y0

    t_fin = (vy0 + np.sqrt(vy0**2 + 2 * g * y0)) / g
    t = np.linspace(0, t_fin, 300)

    x = vx0 * t + x0
    y = -0.5 * g * t**2 + vy0 * t + y0

    fig, ax = plt.subplots(figsize=(8, 5))
    ax.plot(x, y, color="blue", label=f"y = {a:.4f}x² + {b:.2f}x + {c:.2f}")
    ax.axhline(0, color="black", linestyle="--", linewidth=1, alpha=0.7)
    ax.axvline(0, color="black", linestyle="--", linewidth=1, alpha=0.7)

    ax.set_title("Trajectoire de la balle")
    ax.set_xlabel("Distance (m)")
    ax.set_ylabel("Hauteur (m)")
    ax.grid(True)
    ax.legend()

    plt.show()

trajectoire_balle(vx0=25, vy0=15)
```

## **Exercice/ Méthode 107 — Atteindre un point à distance \(L\)**

### **1. Équation de la trajectoire en fonction de \(\alpha\)**

$$
z(x) = -\frac{g}{2 v_0^2 \cos^2 \alpha} x^2 + (\tan \alpha) x
$$

---

### **2. Résolution pour la distance \(L\)**

$$
\sin(2\alpha) = \frac{g L}{v_0^2}
$$

- \(\alpha*1 = \frac{1}{2} \arcsin\left(\frac{g L}{v_0^2}\right)\) *(tir tendu)\_
- \(\alpha*2 = \frac{\pi}{2} - \alpha_1\) *(tir en cloche)\_

---

### **3 & 4. Fonctions Python et tracé**

```python
import numpy as np
import matplotlib.pyplot as plt

def angles_solutions(L, v0=30.0, g=9.81):
    arg = (g * L) / (v0**2)
    if arg > 1.0:
        return None, None
    a1 = 0.5 * np.arcsin(arg)
    a2 = np.pi / 2 - a1
    return np.degrees(a1), np.degrees(a2)

v0 = 30.0
g = 9.81
L_max = (v0**2) / g

L_vals = np.linspace(0, L_max, 500)
a1_vals = []
a2_vals = []

for L in L_vals:
    a1, a2 = angles_solutions(L, v0, g)
    a1_vals.append(a1)
    a2_vals.append(a2)

fig, ax = plt.subplots(figsize=(8, 5))
ax.plot(L_vals, a1_vals, color="blue", label="Tir tendu (α₁)")
ax.plot(L_vals, a2_vals, color="navy", label="Tir en cloche (α₂)")

ax.set_title(f"Angles de tir en fonction de la distance (V₀ = {v0} m/s)")
ax.set_xlabel("Distance L (m)")
ax.set_ylabel("Angle (degrés)")
ax.grid(True)
ax.legend()

plt.show()
```
